PaySME SDK Demo v1.11.1

Test both static and dynamic integration modes

1. Static Mode — Fixed Amount

The merchant pre-defines the amount, invoice ID, and product. The modal opens with these values pre-filled.

<script defer src="https://www.paysme.site/sdk/v1/paysme.js"></script>
<script>
  document.getElementById('pay-btn').addEventListener('click', function() {
    window.PaySME.init({
      vendor_uuid: "32f44735-f665-4b5f-94ca-19d1a45d2942", // Merchant ID UUID; never Vendor ID
      api_key: "YOUR_API_KEY",
      amount_nad: 100,
      invoice_id: "100_Water",
      product_name: "Water Bill",
      recurring: true,
      recurring_period: "monthly",
      on_success: function(result) {
        console.log("Payment code generated:", result.generated_code);
      },
      on_error: function(error) {
        console.error("Payment failed:", error);
      },
      on_cancel: function() {
        console.log("Payment cancelled");
      }
    });
  });
</script>

2. Dynamic Mode — User Selects Amount

The customer chooses the amount and product at payment time. Great for donation pages, custom invoicing, etc.

<script defer src="https://www.paysme.site/sdk/v1/paysme.js"></script>
<script>
  var paysme = window.PaySME.create({
    vendor_uuid: "32f44735-f665-4b5f-94ca-19d1a45d2942", // Merchant ID UUID; never Vendor ID
    api_key: "YOUR_API_KEY",
    on_success: function(result) {
      console.log("Payment code:", result.generated_code);
    },
    on_error: function(error) {
      console.error("Error:", error);
    },
    on_cancel: function() {
      console.log("Cancelled");
    }
  });

  document.getElementById('pay-btn').addEventListener('click', function() {
    paysme.open();
  });
</script>

3. Pending Pay Mode — Merchant Passes the Code

The merchant already has the generated_code from when it was created. They pass it directly — the modal validates the code and shows payment options (Copy, IPP, Card). No forms, no lookups by email/mobile.

<script defer src="https://www.paysme.site/sdk/v1/paysme.js"></script>
<script>
  // Merchant already has the generated_code from when the paycode was created
  var pendingPay = window.PaySME.pending({
    vendor_uuid: "32f44735-f665-4b5f-94ca-19d1a45d2942", // Merchant ID UUID; never Vendor ID
    api_key: "YOUR_API_KEY",
    generated_code: "1234-5678-9012",  // required — the pending paycode
    on_success: function(result) {
      console.log("Code validated:", result.generated_code);
    },
    on_error: function(error) {
      console.error("Error:", error);
    },
    on_cancel: function() {
      console.log("Modal closed");
    }
  });

  document.getElementById('pending-btn').addEventListener('click', function() {
    pendingPay.open();
  });
</script>
